  6def AIRM(A:np.ndarray, B: np.ndarray)->float:
  7    """
  8    Computes the Affine Invariant Riemannian Metric (AIRM) of two Symmetric Positive Definite (SPD) matrices A and B. 
  9
 10    The function uses the generalized eigenvalue problem to compute the AIRM.
 11
 12    Note that for efficiency reasons it is not checked if the inputs are positive definite. 
 13    
 14    --------------------------------------------------------------------------------------
 15
 16    Parameters:
 17
 18    A (np.array): First matrix
 19
 20    B (np.array): Second matrix
 21    
 22    Returns:
 23
 24    float: AIRM of A and B. 
 25    """
 26    assert isinstance(A, np.ndarray), "A should be a numpy array"
 27    assert isinstance(B, np.ndarray), "B should be a numpy array"
 28    assert A.shape == B.shape, "A and B should have the same shape"
 29    assert len(A.shape) == 2 and len(A.shape) == 2, "A and B should be 2D matrices"
 30    assert A.shape[0] == A.shape[1], "A should be a square matrix"
 31    assert B.shape[0] == B.shape[1], "B should be a square matrix"
 32    assert np.allclose(A, A.T), "A should be a symmetric matrix"
 33    assert np.allclose(B, B.T), "B should be a symmetric matrix"
 34
 35    
 36    def get_eigenvalues(A,B):
 37        """
 38        Gets generalized eigenvalues of A and B.
 39        """
 40        return sp.linalg.eigh(A, B, eigvals_only=True)
 41    
 42
 43    return np.sqrt(np.sum(np.power(np.log(get_eigenvalues(A,B)),2)))

 46def get_K_eigenbasis_matrix(N: int) -> np.ndarray:
 47    """
 48    Creates a orthogonal matrix with ON eigenvectors of a complete graph with N nodes.
 49    
 50    Parameters:
 51
 52    N (int): Number of nodes
 53    
 54    Returns:
 55
 56    np.array: Orthogonal matrix with ON eigenvectors of a complete graph with N nodes. 
 57    
 58    """
 59    assert isinstance(N, int), "N should be an integer"
 60    assert N > 0, "N should be a positive integer"
 61
 62    S = np.zeros((N, N))
 63    S[:, 0] = 1
 64    S[:, 0] = normalize_vector(S[:, 0])
 65
 66    for i in range(N-1):
 67        S[i, i+1] = i+1-N
 68        S[i+1:, i+1] = 1
 69        S[:, i+1] = normalize_vector(S[:, i+1])
 70
 71    return S

406def make_GRAMPA_similarity_matrix(A:np.ndarray, B:np.ndarray, bandwith:float) -> np.ndarray:
407    """
408    Creates a similarity matrix of matrices A and B for GRAMPA algorithm to be further rounded to a permutation matrix.   
409    
410    Parameters:
411
412    A (np.array): Adjacency matrix or normalized adjacency matrix of the first graph
413    B (np.array): Adjacency matrix or normalized adjacency matrix of the second graph
414    bandwith (float): Bandwith parameter for the Cauchy kernel
415    
416    Returns:
417
418    np.array: Similarity matrix for GRAMPA algorithm to be rounded to a permutation matrix.
419    
420    VERY INEFFICIENT!!!
421    """
422
423    assert isinstance(A, np.ndarray), "A should be a numpy array"
424    assert isinstance(B, np.ndarray), "B should be a numpy array"
425    assert len(A.shape) == 2, "A should be a 2D matrix"
426    assert len(B.shape) == 2, "B should be a 2D matrix"
427    assert A.shape[0] == A.shape[1], "A should be a square matrix"
428    assert B.shape[0] == B.shape[1], "B should be a square matrix"
429    assert A.shape[0] == B.shape[0], "A and B should have the same size"
430    assert isinstance(bandwith, float), "bandwith should be a float"
431    assert bandwith > 0, "bandwith should be positive"
432    assert np.allclose(A, A.T), "A should be a symmetric matrix"
433    assert np.allclose(B, B.T), "B should be a symmetric matrix"
434
435    N = A.shape[0]
436   
437    def cauchy_kernel(x:float, y:float, bandwith:float) -> float:
438        """
439        Cauchy kernel function.
440        """
441        return 1/(bandwith**2 + (x - y)**2)
442    
443    eigenvalues_A, eigenvectors_A = np.linalg.eigh(A)
444    eigenvalues_B, eigenvectors_B = np.linalg.eigh(B)
445
446    X = np.zeros((N,N))
447    for i in range(N):
448        for j in range(N):
449            X += cauchy_kernel(eigenvalues_A[i], eigenvalues_B[j], bandwith) \
450                * np.outer(eigenvectors_A[:,i], eigenvectors_B[:,j]) * np.sum(eigenvectors_A[:,i]) * np.sum(eigenvectors_B[:,j]) 
451
452    return X

455def make_GRAMPA_similarity_matrix_vectorized(A: np.ndarray, B: np.ndarray, bandwidth: float) -> np.ndarray:
456    """
457    Creates a similarity matrix of matrices A and B for GRAMPA algorithm to be further rounded to a permutation matrix.   
458    
459    Parameters:
460
461    A (np.array): Adjacency matrix or normalized adjacency matrix of the first graph
462    B (np.array): Adjacency matrix or normalized adjacency matrix of the second graph
463    bandwith (float): Bandwith parameter for the Cauchy kernel
464    
465    Returns:
466
467    np.array: Similarity matrix for GRAMPA algorithm to be rounded to a permutation matrix.
468    
469    !VIBECODED!
470    """
471
472    assert isinstance(A, np.ndarray), "A should be a numpy array"
473    assert isinstance(B, np.ndarray), "B should be a numpy array"
474    assert len(A.shape) == 2, "A should be a 2D matrix"
475    assert len(B.shape) == 2, "B should be a 2D matrix"
476    assert A.shape[0] == A.shape[1], "A should be a square matrix"
477    assert B.shape[0] == B.shape[1], "B should be a square matrix"
478    assert A.shape[0] == B.shape[0], "A and B should have the same size"
479    assert isinstance(bandwidth, float), "bandwith should be a float"
480    assert bandwidth > 0, "bandwith should be positive"
481    assert np.allclose(A, A.T), "A should be a symmetric matrix"
482    assert np.allclose(B, B.T), "B should be a symmetric matrix"
483
484    def cauchy_kernel_matrix(evals_A, evals_B, bandwidth):
485        # evals_A, evals_B: shape (N,)
486        diff = evals_A[:, None] - evals_B[None, :]      # (N, N)
487        return 1.0 / (bandwidth**2 + diff**2)           # (N, N)
488
489        N = A.shape[0]
490
491    # Eigen-decompositions
492    evals_A, evecs_A = np.linalg.eigh(A)            # (N,), (N, N)
493    evals_B, evecs_B = np.linalg.eigh(B)            # (N,), (N, N)
494
495    # Cauchy kernel on eigenvalues
496    K = cauchy_kernel_matrix(evals_A, evals_B, bandwidth)   # (N, N)
497
498    # Sums of eigenvectors (per eigenvector)
499    sA = evecs_A.sum(axis=0)                        # (N,)
500    sB = evecs_B.sum(axis=0)                        # (N,)
501
502    # Combine scalar weights into a single weight matrix W_ij
503    W = K * (sA[:, None] * sB[None, :])             # (N, N)
504
505    # Vectorized sum over outer products:
506    # X_{pq} = sum_{i,j} W_ij * evecs_A[p,i] * evecs_B[q,j]
507    # This is a double contraction that can be done with einsum.
508    X = np.einsum('ij,pi,qj->pq', W, evecs_A, evecs_B, optimize=True)  # (N, N)
509
510    return X

513def hungarian_algorithm(X: np.ndarray) -> np.ndarray:
514    """
515    Given a matrix X (n x n),
516    return a permutation p as a 1D array of length n such that
517    p[i] is the column index assigned to row i, maximizing sum |X[i, p[i]]|.
518    !!!VIBECODED!!!
519    """
520    row_ind, col_ind = sp.optimize.linear_sum_assignment(-X)  # Hungarian algorithm[web:31]
521
522    # row_ind is sorted, so this is already in row order
523    # permutation p: maps row i -> column p[i]
524
525    return col_ind
